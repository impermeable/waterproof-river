\begin{theorem}[Leibniz Test, a.k.a. Alternating Series Test]
	Let $a, b : \mathbb{N} \to \mathbb{R}$ be two real-valued sequence such that for all $k \in \mathbb{N}$, $b_k = (-1)^k a_k$.
	Assume that there exists a $K \in \mathbb{N}$ such that 
	\begin{enumerate}
		\item $a_k \geq 0$ for every $k \geq K$
		\item $a_k \geq a_{k+1}$ for every $k \geq K$
		\item $\lim_{k \to \infty} a_k = 0$	
	\end{enumerate}	
	Then, the series
	\[
	\sum_{k = K}^\infty b_k = \sum_{k = K}^\infty (-1)^k a_k
	\]
	is convergent. In addition, the following estimate holds for every $N \geq K$,
	\[
	\left| S_N - \sum_{k=K}^\infty b_k \right| < a_{N+1}
	\]
	where for all $n \in \mathbb{N}$, $S_n := \sum_{k=K}^N b_k$.
\end{theorem}

\begin{proof}
	We only prove the case in which $K = 0$.
	
	We note that $(S_{2n})$ is a decreasing sequence, because for all $n \in \mathbb{N}$,
	\[
	S_{2n + 2} = S_{2n} - a_{2n+1} + a_{2n+2} \leq S_{2n}.
	\]
	Similarly, $(S_{2n+1})$ is an increasing sequence, because for all $n \in \mathbb{N}$,
	\[
	S_{2n+3} = S_{2n+1} + a_{2n+2} - a_{2n+3} \geq S_{2n+1}.
	\]
	Finally, we note that for all $n \in \mathbb{N}$, it holds that
	\[
	S_1 \leq S_{2n + 1} = S_{2n} - a_{2n+1} \leq S_{2n} \leq S_0.
	\]
	As a consequence, the sequence $(S_{2n})$ is bounded from below by $S_1$, and the sequence $(S_{2n+1})$ is bounded above by $S_0$. Therefore, both sequences are convergent.
	
	Because the sequence $(a_n)$ converges to zero, we can show that the sequence $(a_{2n+1})$ converges to zero as well.
	By the limit laws, we find that 
	\[
	\lim_{n \to \infty} S_{2n+1} = \lim_{n \to \infty} S_{2n} - \lim_{n \to \infty} a_{2n+1} = \lim_{n \to \infty} S_{2n}.
	\]
	In words, the sequences $(S_{2n+1})$ and $(S_{2n})$ converge to the same limit. Let's call this limit $s$.
	
	Finally, 
	\[
	S_{2n} > s > S_{2n+1} = S_{2n} - a_{2n+1}
	\]
	so that $|S_{2n} - s| < a_{2n+1}$.
	
	Similarly,
	\[
	S_{2n+1} < s < S_{2n+2} = S_{2n+1} + a_{2n+2}
	\]
	so that $|S_{2n+1} - s| < a_{2n+2}$.
	
	In conclusion, in such an alternating series we have the estimate for all $n \in \mathbb{N}$
	\[
	|S_n - s| \leq a_{n+1}
	\]
	Therefore the sequence $(S_n)$ converges to $s$.
\end{proof}

\begin{example}
	We claim that the series
	\[
	\sum_{k=1}^\infty (-1)^k \frac{1}{k}
	\]
	converges.

	We would like to apply the Alternating Series Test. To do so, we need to check its conditions.
	
	We define the sequence $a : \mathbb{N} \to \mathbb{R}$ by
	\[
	a_k := \frac{1}{k}
	\]
	for $k \geq 1$ (and $a_0 = a_1 = 1$).
	
	We now check the conditions for the Alternating Series Test.
	\begin{enumerate}
		\item We need to show that $a_k \geq 0$ for all $k \in \mathbb{N}$. Let $k \in \mathbb{N}$. Then
		\[
		a_k = \frac{1}{k} \geq 0.
		\]
		\item We need to show that $a_k \geq a_{k+1}$ for all $k \in \mathbb{N}$. Let $k \in \mathbb{N}$. Then
		\[
		a_k = \frac{1}{k} \geq \frac{1}{k+1} = a_{k+1}.
		\]
		\item We need to show that 
		\[
		\lim_{k \to \infty} a_k = 0.
		\]
		This follows as this is a standard limit.
	\end{enumerate}
	It follows from the Alternating Series Test that the series
	\[
	\sum_{k=1}^\infty (-1)^k a_k = \sum_{k=1}^\infty (-1)^k \frac{1}{k}
	\]
	converges.
\end{example}
