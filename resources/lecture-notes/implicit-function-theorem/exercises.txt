\begin{orange_exercise}
\label{ex:1-plus-1-implicit}
Consider the function $F: \mathbb{R}^2 \to \mathbb{R}$ given by
\[
F(x, y) = x^2 - y^3 + 3 y
\]	
\begin{enumerate}
		\renewcommand{\theenumi}{\alph{enumi}}
\item Specify precisely the points $(a, b)$ on the curve
\[
\Gamma := \{ (x, y) \in \mathbb{R}^2 \ | \  F(x, y ) = 1 \}
\]
such that there {\bf does not} exist $r_1, r_2> 0$ and a continuously differentiable function $g: B(a, r_1) \to \mathbb{R}$ such that for all $x \in B(a, r_1)$ and $y \in B(b, r_2)$
\[
F(x, y) = 1 \qquad \text{ if and only if } \qquad y = g(x).
\]
\item For a continuously differentiable function $g : B(a, r_1) \to \mathbb{R}$ such that
\[
F(x, g(x)) = 1
\]
compute $g'(x)$ for every $x \in B(a, r_1)$ in terms of $x$ and $g(x)$.
\end{enumerate}
\end{orange_exercise}

\begin{orange_exercise}
\label{ex:characteristics-implicit}
Let $\rho : \mathbb{R} \to \mathbb{R}$ be a continuously differentiable function, and consider the map $F: \mathbb{R}^2 \to \mathbb{R}^2$ given by
\[
F( x, t ) := (x + \rho(x) t, t)
\]
Show that there exist radii $r_1 > 0$ and $r_2 > 0$ and a function $G : B(0, r_2) \to B(0, r_1)$ such that for all $(y,s)$ in $B(0,r_2)$ and $(x,t)$ in $B(0,r_1)$
\[
(y, s) = F((x, t)) \text{ if and only if } (x, t) = G((y, s)).
\]
and for every $(y, s)$ give an expression for
\[
[DG]_{(y, s)}
\]
\end{orange_exercise}

\begin{restatable}{orange_exercise}{reStandardRImplicit}
Consider the function $F : \mathbb{R}^3 \times \mathbb{R}^2 \to \mathbb{R}^2$	given by
\[
F(x_1, x_2, x_3, y_1, y_2) = ((x_2)^2y_1 + y_2 - 3, \cos(x_3 y_2) + y_2 + (x_1)^2)
\]
Prove that there exist $r_1, r_2 > 0$ and a continuously differentiable function $g : B((1, 3, 2), r_1) \to \mathbb{R}^2$ such that for all $x \in B((1, 3, 2), r_1)$ and all $y \in B((1, 0), r_2)$,
\[
F(x_1, x_2, x_3, y_1, y_2) = (6, 2) \quad \text{ if and only if } \quad y = g(x).
\]
Moreover, compute the Jacobian $[Dg]_{(1,3,2)}$.
\end{restatable}

\begin{restatable}{orange_exercise}{reStandardRImplicitPractice}
Consider the function $f : \mathbb{R}^3 \to \mathbb{R}$ defined by
\[  f((x_1, x_2, y_1)) =  \exp( x_2 \sin( y_1 )) - 2y_1 - (x_1)^2 + (x_2)^4 \]
\begin{enumerate}
	\renewcommand{\theenumi}{\alph{enumi}}
	\item Prove that there exist $r_1 > 0$ and $r_2 > 0$ and a function $g : B( (2, 1), r_1) \to \mathbb{R}$ such that for all $(x_1, x_2) \in B((2, 1) , r_1)$, and $y_1 \in ( - r_2, r_2)$,
	\[ f((x_1, x_2, y_1)) = -2 \qquad \text{ if and only if } \qquad y_1 = g((x_1, x_2)). \]
	\item Compute $\nabla g$ in the point $(2, 1)$.
\end{enumerate}

\end{restatable}
