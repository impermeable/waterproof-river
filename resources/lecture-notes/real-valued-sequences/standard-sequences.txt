\subsection{Geometric sequence}

\begin{proposition}[{Standard limit of geometric sequence}]
	\label{pr:behavior-q-to-power-n}
	Let $q \in \mathbb{R}$.
	The sequence $(a_n)$ defined by $a_n := q^n$ for $n \in \mathbb{N}$
	\begin{itemize}
		\item converges to $0$ if $q \in (-1,1)$
		\item converges to $1$ if $q = 1$
		\item diverges to $\infty$ if $q > 1$
		\item diverges, but not to $\infty$ or $-\infty$ if $q \leq -1$.
	\end{itemize}
\end{proposition}
\begin{proof}
	If $q = 0$ then it is clear that the sequence $n \mapsto q^n$ converges to $0$. If $q = 1$ then it is clear that the sequence $n \mapsto q^n$ converges to $1$. 
	
	If $q \in (0,1)$, then the sequence $(a_n)$ is decreasing and bounded from below by $0$. Therefore, the sequence $(a_n)$ is convergent. Moreover,
	
	\[
	s = \lim_{ n \to \infty} a_{n+1} = q \lim_{n \to \infty} a_n = q s
	\]
	so $s = 0$.
	
	If $q \in (-1,0)$, then
	\[
	- |q|^n \leq q^n \leq |q|^n
	\]
	and it follows from the squeeze theorem and the previous part of the proof that $\lim_{n \to \infty} q^n = 0$.
	
	Now assume $q > 1$. We will show that the sequence $n \mapsto q^n$ diverges to $\infty$. Let $M \in \mathbb{R}$. 
	
	Then we can write $q := 1 + b$ for some $b > 0$. 
	
	Choose $N := M \lceil 1/b \rceil$. Let $n \geq N$. By the Bernoulli inequality, it follows that
	\[
	q^n = (1 + b)^n \geq 1 + n b \geq 1 + N b \geq 1 + M \frac{1}{b}b > M.
	\]	
	Finally, we consider the case $q \leq - 1$. Suppose the sequence $n \mapsto (-q)^n$ converges to some $r \geq 0$. Then we may choose an $N \in \mathbb{N}$ such that for all $n \geq N$, $|(-q)^n - r | < 1/2$. Therefore, 
	\[
	r - (-q)^{2N + 1} < 1/2
	\]
	which is a contradiction because $(-q)^{2N + 1} = -q^{2N + 1} \leq -1$. 
	
	In a similar way, we can rule out that $n \mapsto (-q)^n$ converges to some $r < 0$, or diverges to $\infty$ or diverges to $-\infty$.
\end{proof}

\subsection{The \texorpdfstring{$n$}{n}th root of \texorpdfstring{$n$}{n}}

\begin{proposition}[{Standard limit of $(\sqrt[n]{n})$}]
	The sequence $(b_n)$ defined as $b_n := \sqrt[n]{n}$ converges to $1$.
\end{proposition}
\begin{proof}
	We write $b_n = 1 + d_n$, where $d_n \geq 0$. By the limit theorems, it suffices to show that $\lim_{n \to \infty} d_n = 0$. Note that $b_n^n = (1 + d_n)^n = n$.
	Let $n \geq 2$. Then
	\[
	1 + {n \choose 2} d_n^2 \leq \sum_{k=0}^n {n \choose k} d_n^k =(1 + d_n)^n = n.
	\]
	Therefore,
	\[
	0 < d_n^2 \leq \frac{2}{n}
	\]
	and
	\[
	0 < d_n \leq \sqrt{\frac{2}{n}}.
	\]
	The limit of the left-hand side is zero, and by the limit theorems, we know that the limit of the right-hand-side is $0$ as well. It follows by the squeeze theorem that $\lim_{n \to \infty} d_n = 0$. 
\end{proof}

\begin{corollary}
	Let $a > 0$. Then the sequence $(b_n)$ defined by $b_n := \sqrt[n]{a}$ converges to $1$.
\end{corollary}

\subsection{The number \texorpdfstring{$e$}{e}}
\label{se:number-e}

In this chapter we are going to introduce the number $e$ as follows. We will first define the sequence
\[
a_n := \left(1 + \frac{1}{n} \right)^n
\]
We will show that this sequence is bounded and increasing. It therefore has a limit value, and that limit value is called $e$.

\begin{lemma}
	The sequence $(a_n)$ defined by $a_n := (1 + 1/n)^n$ for $n \in \mathbb{N} \setminus \{0\}$ and $a_0=1$ is strictly increasing.
\end{lemma}
\begin{proof}
	We need to show that for all $n \in \mathbb{N}\setminus \{0\}$, $a_n < a_{n+1}$. Let $n \in \mathbb{N}$ be larger than or equal to $1$.  
	We can just write out
	\[
	\begin{split}
		a_n &= \sum_{k=0}^n {n \choose k} \left(\frac{1}{n}\right)^k = \sum_{k=0}^n \frac{1}{k!} \frac{n!}{(n-k)!} \left(\frac{1}{n} \right)^k
	\end{split}
	\]	
	whereas
	\[
	\begin{split}
		a_{n+1} &= \sum_{k=0}^{n+1} \frac{1}{k!} \frac{(n+1)!}{(n+1 - k)!} \left(\frac{1}{n+1} \right)^{k}
	\end{split}
	\]
	How to compare these and show that $a_n < a_{n+1}$? First, because all terms in the sum are positive, we can estimate $a_{n+1}$ from below by forgetting the last term
	\[
	\begin{split}
		a_{n+1} &> \sum_{k=0}^{n} \frac{1}{k!} \frac{(n+1)!}{(n+1 - k)!} \left(\frac{1}{n+1} \right)^{k}
	\end{split}
	\]
	Next, we will show that each of the term in this sum is larger than the corresponding sum for $a_n$. We can see this better if we rewrite
	\[
	\begin{split}
		a_{n+1} &> \sum_{k=0}^n \frac{1}{k!} \left(\frac{n+1}{n+1}\right) \left(\frac{n+1-1}{n+1}\right) 
		\cdots \left(\frac{n+1 - (k-1)}{n+1} \right) \\
		&> \sum_{k=0}^{n} \frac{1}{k!} \left(\frac{n}{n}\right) \left( \frac{n-1}{n}\right)\cdots \left(\frac{n-(k-1)}{n} \right)\\
		&= a_n
	\end{split}
	\]
\end{proof}

\begin{lemma}
	The sequence $(a_n)$ defined by $a_n = (1 + 1/n)^n$ for $n \in \mathbb{N} \setminus \{0\}$ (and $a_0 = 1$) is bounded from above by $3$. 	
\end{lemma}
\begin{proof}
	Again we write
	\[
	\begin{split}
		a_n &= \sum_{k=0}^n {n \choose k} \left(\frac{1}{n} \right)^k \\
		&= \sum_{k=0}^n \frac{1}{k!} \left(\frac{n}{n}\right) \left( \frac{n-1}{n}\right)\cdots \left(\frac{n-(k-1)}{n} \right) \\
		&\leq \sum_{k=0}^n \frac{1}{k!} = 1 + \sum_{k=1}^n \frac{1}{k!} \leq 1 + \sum_{k=1}^n \frac{1}{2^{k-1}}\\
		&\leq 1 + 2 = 3.
	\end{split}
	\]
\end{proof}

By the previous lemmas, the sequence
\[
n \mapsto \left(1 + \frac{1}{n} \right)^n
\]
converges. Let's record in the next definition that we call the limit $e$.

\begin{definition}[{Standard limit corresponding to the number $e$}]
We define
\[
e := \lim_{n \to \infty} \left(1 + \frac{1}{n} \right)^{n}
\]	
\end{definition}

\subsection{Exponentials beat powers}

\begin{proposition}
	Let $a \in (1, \infty)$ and let $p \in (0,\infty)$. Then
	\[
	\lim_{n \to \infty} \frac{n^p}{a^n} = 0.
	\]	
\end{proposition}

\begin{proof}
Define $b := a - 1 > 0$, so that $a = 1 + b$. By the Archimedean property there exists an $M \in \mathbb{N}$ such that $M > p+1$.	Define $N := 2 M$.

We now claim that for all $n \geq N$, 
\[
a^n \geq \frac{n^M}{2^M M!} b^M.
\]
Indeed, let $n \geq N$. First note that because $n \geq 2M$, we know
\begin{equation}
	\label{eq:exp-beat-powers-aux}
n - M \geq \frac{n}{2}.
\end{equation}
We now compute
\[
\begin{split}
a^n = (1 + b)^n &= \sum_{k=0}^n {n \choose k} b^k\\
&\geq {n \choose M} b^M \\
&= n (n-1) \cdots (n - M + 1) \frac{1}{M!} b^M\\
&\geq \left(\frac{n}{2} \right)^M \frac{1}{M!} b^M
\end{split}
\]
where for the last inequality we used (\ref{eq:exp-beat-powers-aux}).
This proves our claim.

Since $M > p + 1$ we find that for all $n \geq N$
\[
0 < \frac{n^p}{a^n} < 2^M M! \frac{1}{b^M} \frac{1}{n}.
\]
We know that
\[
\lim_{n \to \infty} 2^M M! \frac{1}{b^M} \frac{1}{n} = 0
\]
by limit theorems and the standard limit $\lim_{n \to \infty} \frac{1}{n} = 0$. 
Therefore, it holds by the squeeze theorem (Theorem \ref{th:squeeze-theorem}) that
\[
\lim_{n \to \infty} \frac{n^p}{a^n} = 0.
\]
\end{proof}

\section*{Sequences with values in $\mathbb{R}^d$}

(Note: this topic originally occurred further down the lecture notes, but I have moved it forward so we may get more concrete examples of sequences)

\begin{proposition}
	\label{pr:seq-converges-comp-converge}
	Consider the metric space $(\mathbb{R}^d, \| \cdot \|_2)$. Let $z \in \mathbb{R}^d$ and let $x : \mathbb{N} \to \mathbb{R}^d$ be a sequence (we are going to denote this sequence also as $(x^{(n)})$).
	Denote by $y_i$ the $i$th component of a vector $y \in \mathbb{R}^d$. 
	Then the sequence $(x^{(n)})$ converges to $z$ if and only if for all $i \in \{ 1, \dots, d\}$, the sequence $(x^{(n)}_i)$ converges to $z_i$.
\end{proposition}

\begin{example}
Consider the sequence $x : \mathbb{N} \to \mathbb{R}^2$	taking values in the normed vector space $(\mathbb{R}^d, \| \cdot \|_2)$, defined by
\[
x^{(n)} := \left(\frac{1}{n}  , \left(\frac{1}{2}\right)^n \right)
\]
for $n \geq 1$. We use a \emph{super}script for the index ($n$) of the sequence, so that we can use \emph{sub}scripts for the components of the sequence, i.e. the first component sequence $(a^{(n)}_1)$ is given by
\[
x^{(n)}_1 = \frac{1}{n}
\]
and the second component sequence $(a^{(n)}_2)$ is given by 
\[
x^{(n)}_2 = \left(\frac{1}{2}\right)^n
\]
By standard limits, we know that both
\[
\lim_{n \to \infty} x^{(n)}_1 = \lim_{n \to \infty} \frac{1}{n} = 0
\]
and
\[
\lim_{n \to \infty} x^{(n)}_2 = \lim_{n \to \infty} \left(\frac{1}{2} \right)^n = 0.
\]
By Proposition \ref{pr:seq-converges-comp-converge} it follows that the sequence $x: \mathbb{N} \to \mathbb{R}^2$ converges to
\[
\left(\lim_{n \to \infty} \frac{1}{n}, \lim_{n \to \infty} \left(\frac{1}{2} \right)^n \right) = (0, 0) = 0.
\]
Note how in the last term we use the notation $0$ for the $0$-vector in the vector space $\mathbb{R}^2$.
\end{example}
