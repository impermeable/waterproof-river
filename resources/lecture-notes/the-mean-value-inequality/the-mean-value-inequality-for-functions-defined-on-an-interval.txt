\begin{lemma}[Mean-value inequality (v0)]
	\label{le:MVI}
	Let $f : [a, b] \to W$ be continuous on $[a, b]$ and differentiable on $(a, b)$. Then
	\[
	\|f(b) - f(a)\|_W \leq \sup_{t \in (a, b)} \|f'(t) \|_W (b - a).
	\]
\end{lemma}

\begin{proof}
	Denote
	\[
	K := \sup_{ t \in (a, b) } \| f'(t) \|_W.
	\]
	
	In this first part of the proof we will a few times use an "It suffices to show that-" construction to reduce what we need to show to an easier statement.
	
	We first claim that it suffices to show that for all $\bar{a} \in (a, b)$
	\begin{equation}
		\label{eq:inequality-with-bar-a-b}
	\| f(b) - f(\bar{a})\|_W \leq K (b - \bar{a}).
	\end{equation}
	To see why this suffices, note that the left-hand side and right-hand side can be viewed as continuous functions of $\bar{a}$. Therefore, if we know that (\ref{eq:inequality-with-bar-a-b}) holds, we can take the limit as $\bar{a} \to a$ on both sides, and conclude that also
	\[
	\| f(b) - f(a) \|_W \leq K ( b - a).
	\]
	
	Let therefore $\bar{a} \in (a, b)$. We aim to show (\ref{eq:inequality-with-bar-a-b}).
	
	We now claim that it suffices to show that for all $\epsilon > 0$, all $s \in [\bar{a}, b]$,
	\begin{equation}
		\label{eq:inequality-in-mv-inequality}
		\|f(s) - f(\bar{a})\|_W \leq (K + \epsilon) (s - \bar{a}).
	\end{equation}

	To prove the claim, we aim to show (\ref{eq:inequality-with-bar-a-b}) from (\ref{eq:inequality-in-mv-inequality}). First note that if (\ref{eq:inequality-in-mv-inequality}) holds for all $\epsilon>0$ and all $s \in [\bar{a}, b]$, then it also holds for all $\epsilon > 0$ and $s = b$. We now argue by contradiction. Suppose
	\[
	\| f(b) - f(\bar{a}) \|_W > K(b - \bar{a})
	\]
	Then, we may define
	\[
	\epsilon_1 := \frac{1}{2}\left(\frac{\| f(b) - f(\bar{a})\|_W}{b - \bar{a}} - K\right) > 0
	\]
	so that
	\[
	\|f(b) - f(\bar{a}) \|_W > (K + \epsilon_1) (b - \bar{a}).
	\]
	Yet when we choose $\epsilon = \epsilon_1$ in (\ref{eq:inequality-in-mv-inequality}), we obtain
	\[
	\|f(b) - f(\bar{a}) \|_W \leq (K + \epsilon_1) (b - \bar{a}).
	\]
	which is a contradiction. We conclude that (\ref{eq:inequality-with-bar-a-b}) holds.
	
	After all these reductions, we are left with showing that for all $\epsilon > 0$ and all $s \in [\bar{a}, b]$, indeed (\ref{eq:inequality-in-mv-inequality}) holds.
	
	Let $\epsilon > 0$.
	
	We will now show three claims.
	
	Our first claim is that inequality (\ref{eq:inequality-in-mv-inequality}) holds for $s = \bar{a}$. This holds because
	\[
	\| f(a) - f(\bar{a}) \|_W = 0 \leq 0 = (K+\epsilon) (\bar{a} - \bar{a}).
	\]
	
	Our second claim is that whenever inequality (\ref{eq:inequality-in-mv-inequality}) holds for all $s \in [\bar{a}, c)$ for some $c \in [\bar{a}, b]$, it also holds for $s = c$. This follows since the left-hand side and the right-hand of the inequality (\ref{eq:inequality-in-mv-inequality}) are continuous when interpreted as functions in $s$.
	
	Our third claim is that whenever the inequality holds for all $s \in [\bar{a}, c]$ for some $c \in [\bar{a}, b)$, there exists a $\delta > 0$ such that the inequality holds for all $s \in [\bar{a}, c + \delta)$.
	
	To prove this third claim, let $c \in [\bar{a}, b)$ and assume the inequality holds for all $s \in [\bar{a}, c]$. Since $f$ is differentiable in $c$, there exists a $\delta > 0$ such that for all $s \in [c, c + \delta)$,
	\[
	\|f(s) - f(c) - f'(c)(s - c) \|_W = \| \mathsf{Err}_c^f ( s ) \|_W \leq \epsilon |s - c| = \epsilon (s - c)
	\]
	Therefore by the triangle inequality for all $s \in [c, c + \delta)$,
	\[
	\begin{split}
	\|f(s) - f(c)\|_W &= \| \mathsf{Err}_c^f(s) + f'(c) (s - c) \|_W\\
	&\leq \|f'(c)(s - c) \|_W + \epsilon(s - c) \leq (K + \epsilon) (s - c)
	\end{split}
	\]
	As a consequence,
	\[
	\begin{split}
		\|f(s) - f(\bar{a}) \|_W &\leq \| f(s)- f(c)\|_W + \|f(c) - f(\bar{a})\|_W \\
		&\leq (K + \epsilon)(s - c) + (K+ \epsilon) (c - \bar{a}) = (K + \epsilon) (s - \bar{a}).
	\end{split}
	\]
	which shows that indeed inequality (\ref{eq:inequality-in-mv-inequality}) is also satisfied for all $s \in [\bar{a}, c + \delta)$. Hence we have proved the third claim.
	
	We now define the set $S$ as those $s \in [\bar{a}, b]$ such that for all $\sigma \in [\bar{a}, s]$, inequality (\ref{eq:inequality-in-mv-inequality}) is satisfied. In other words,
	\[
	S := \{s \in [\bar{a}, b] \ | \ \| f(\sigma) - f(\bar{a})\|_W \leq (K + \epsilon) (\sigma - \bar{a}) \text{ for all } \sigma \in [\bar{a}, s] \}.
	\]
	Note that just from its definition, it follows that $S$ is either the empty set, or just the point $\{\bar{a}\}$ or it is an interval that is closed on the left with $\bar{a}$ as the left endpoint. From the first claim, we know that $\bar{a} \in S$, so $S$ is not empty. The second claim tells us that $S$ is closed, so it is either $\{\bar{a}\}$ or it is a closed interval of the form $[\bar{a}, c]$ with $c \in (\bar{a}, b]$. The third claim gives a contradiction when $S = \{ \bar{a}\}$ or $S = [\bar{a}, c]$ with $c < b$. Therefore $S = [\bar{a}, b]$ and inequality (\ref{eq:inequality-in-mv-inequality}) is satisfied in $s = b$.
\end{proof}
