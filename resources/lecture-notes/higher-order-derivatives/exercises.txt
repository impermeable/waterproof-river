The following exercises are mostly meant as practice to get familiar with the concepts and theorems in this chapter. 

\begin{orange_exercise}
	\label{ex:concluding-second-differentiability}
Consider the function $f: \Omega \to \mathbb{R}$ where $\Omega = \mathbb{R}^2$ given by
\[
f((x_1, x_2)) = \exp( (x_1)^2 - (x_2)^3 )
\]
Show that $f$ is twice differentiable on $\mathbb{R}^2$ by going through the following steps:
\begin{enumerate}
	\renewcommand{\theenumi}{\alph{enumi}}
\item Show that the partial derivative functions of $f$, namely
\[
\frac{\partial f}{\partial x_1} : \Omega \to \mathbb{R} \qquad \text{ and } \qquad \frac{\partial f}{\partial x_2} : \Omega \to \mathbb{R}
\] 
 exist and compute them.
\item Show that the second order partial derivative functions
\begin{align*}
 \frac{\partial^2 f}{\partial x_1 \partial x_1} &: \Omega \to \mathbb{R} &  \frac{\partial^2 f}{\partial x_2 \partial x_1}&: \Omega \to \mathbb{R}\\
 \frac{\partial^2 f}{\partial x_1 \partial x_2} &: \Omega \to \mathbb{R} &  \frac{\partial^2 f}{\partial x_2 \partial x_2}&: \Omega \to \mathbb{R}
\end{align*}
exist and compute them.
\item Now show that the second order partial derivatives are continuous and conclude, by quoting the right theorem, that $f$ is twice differentiable.
\end{enumerate}
\end{orange_exercise}

\begin{orange_exercise}
	\label{ex:practice-with-2nd-order-dir-derivs}
Consider the function $f: \mathbb{R}^2 \to \mathbb{R}$ given by
\[
f((x_1, x_2)) = (x_1)^{5} (x_2)^{8}
\]

\begin{enumerate}
	\renewcommand{\theenumi}{\alph{enumi}}
\item For arbitrary $u \in \mathbb{R}^2$, give the function
\[
D_{u} f : \mathbb{R}^2 \to \mathbb{R}
\]

\item For arbitrary $v \in \mathbb{R}^2$, give the function
\[
(D_{v} (D_{u} f) ) : \mathbb{R}^2 \to \mathbb{R}
\]

\item Define the vectors $u := (1, 3)$ and $v := (7, 2)$. Let $a = (1, 1) \in \mathbb{R}^2$. Give
\[
(D^2f)_a(v, u)
\]
\item For the same choice of $u$, $v$ and $a$, also give
\[
(D^2f)_a(u, v)
\]
\end{enumerate}
\end{orange_exercise}

\begin{orange_exercise}
	\label{ex:practice-with-symmetry}
	About a certain function $f : \mathbb{R}^5 \to \mathbb{R}^2$ the following is known in a point $a \in \mathbb{R}^5$. The function is $4$ times differentiable in $a$, and
	\[
	\begin{split}
	(D^4 f)_a(e_2, e_3, e_3, e_5) &= (5, 0)\\
	(D^4 f)_a(e_2, e_3, e_5, e_5) &= (2, 3)\\
	\frac{\partial^4 f}{\partial x_5 \partial x_3 \partial x_3 \partial x_3}(a) &= (0, 1)\\
	\frac{\partial^4 f}{\partial x_5 \partial x_3 \partial x_5 \partial x_3}(a) &= (1, 2)   
	\end{split}
	\] 
	
	Give
	\[
	(D^4f)_a (e_3 - 2 e_2, 6 e_5, e_3 + e_5, e_3).
	\]
\end{orange_exercise}	

\begin{restatable}{orange_exercise}{reComputeWithHigherDerivs}
	About a certain function $f : \mathbb{R}^2 \to \mathbb{R}$ the following is known in a point $a \in \mathbb{R}^2$. The function is $3$ times differentiable in $a$, and 
	\[
	\begin{split}
		(D^3 f)_a(e_1 + e_2, e_1 + e_2, e_1 + e_2) &= 5\\
		(D^3 f)_a(e_1 - e_2, e_1 - e_2, e_1 - e_2) &= 8\\
		\frac{\partial^3 f}{\partial x_1^3} (a) &= 2\\
		\frac{\partial^3 f}{\partial x_1^2 \partial x_2}(a) &= 3
	\end{split}
	\]
	Write for $y \in \mathbb{R}^2$ the expression
	\[
	(D^3 f)_a (y_1 e_1 + y_2 e_2, y_1 e_1 + y_2 e_2, y_1 e_1 + y_2 e_2)
	\]
	as a polynomial in $y$.
\end{restatable}
