\begin{lemma}
	Let $f : \Omega \to W$ be a function defined on an open domain $\Omega \subset V$.
	Let $a \in \Omega$ and assume that $f$ is twice differentiable in $a$. Then for all $u, v \in V$,
	\[
	(D^2 f)_a (u, v) = (D^2 f)_a (v, u).
	\]
\end{lemma}

\begin{proof}
We will show that for all $u, v \in V$, the limit
\begin{equation}
	\label{eq:symmetry-fin-diff-form-t}
\lim_{t \to 0} \frac{1}{t^2} \left(f(a + t u + t v) - f(a + tu) - f(a + t v) + f(a) \right)
\end{equation}
exists and is equal to
\[
(D^2f)_a(u, v)
\]
Note that the expression in (\ref{eq:symmetry-fin-diff-form-t}) remains the same when $u$ and $v$ are interchanged, so that the limit is also equal to
\[
(D^2 f)_a(v, u).
\]
From there, we conclude that $(D^2 f)_a (u, v) = (D^2 f)_a (v, u)$.

Consider now for $s, t$ real numbers (that are small enough so that $a + s u + t v \in \Omega$) the expression
\[
\begin{split}
&\|f(a + s u + t v) - f(a + s u) - f(a + t v) + f(a) - s t (D^2 f)_a(u, v)\|_W\\
&=\| f(a + s u + t v) - s t (D^2 f)_a (u, v) - f(a + s u) - f(a + t v) + f(a) \|_W
\end{split}
\]
By the Mean-Value Inequality, applied to the function 
\[
g(t) := f(a + s u + t v) - s t (D^2 f)_a (u, v) - f(a + t v)
\]
we find that
\begin{equation}
\label{eq:comp-symmetry}
\begin{split}
&\| f(a + s u + t v) - s t (D^2 f)_a (u, v) - f(a + s u) - f(a + t v) + f(a) \|_W\\
&= \|g(t) - g(0) \|_W \\
&\leq \sup_{\tau \in (-|t|, |t|)} \| (Df)_{a + su + \tau v}(v) - s (D^2f)_a (u, v) - (Df)_{a + \tau v}(v) \|_W | t |\\
\end{split}
\end{equation}
We now use the differentiability of the function $(Df) : \Omega \to \Lin(V, W)$ in the point $a$ so that
\[
\begin{split}
(Df)_{a + su + \tau v}
= (Df)_a + (D(Df))_a (su + \tau v) + \mathsf{Err}_a^{Df} (a + su + \tau v)
\end{split}
\]
and 
\[
(Df)_{a + \tau v} = (Df)_a + (D(Df))_{a} (\tau v) + \mathsf{Err}_a^{Df}(a + \tau v)
\]
Note that the left-hand side and right hand side of these equations are linear maps. We apply these linear maps to the vector $v \in V$ and find
\[
\begin{split}
	(Df)_{a + su + \tau v}(v)
	&:= (Df)_a(v) + (D(Df))_a (su + \tau v)(v) \\
	&\qquad + \mathsf{Err}_a^{Df} (a + su + \tau v) (v)\\
	&= (Df_a)(v) + s (D^2 f)_a (u, v) + \tau (D^2 f)_a(v, v) \\
	&\qquad + \mathsf{Err}_a^{Df} (a + su + \tau v) (v)
\end{split}
\]
and 
\[
\begin{split}
(Df)_{a + \tau v}(v) &:= (Df)_a(v) + (D(Df))_{a} (\tau v)(v) + \mathsf{Err}_a^{Df}(a + \tau v)(v)\\
&:= (Df)_a(v) + \tau (D^2f)_a(v, v) + \mathsf{Err}_a^{Df}(a + \tau v)(v).
\end{split}
\]
We now continue from the final line in the computation (\ref{eq:comp-symmetry})
\[
\begin{split}
&\sup_{\tau \in (-|t|, |t|)} \| (Df)_{a + su + \tau v}(v) - s (D^2f)_a (u, v) - (Df)_{a + \tau v}(v) \|_W | t |\\
&=\sup_{\tau \in (-|t|, |t|)} \| \mathsf{Err}_{a}^{Df}(a + su + \tau v)(v)- \mathsf{Err}_a^{Df}(a + \tau v)(v) \|_W |t|\\
&\leq \sup_{\tau \in (-|t|, |t|)} \left( \big\| \mathsf{Err}_{a}^{Df}(a + su + \tau v) \big\|_{V \to W} 
    + \big\|\mathsf{Err}_a^{Df}(a + \tau v) \big\|_{V \to W} \right) |t| \| v \|_V.
\end{split}
\]
It follows that for $t \neq 0$,
\[
\begin{split}
&\left\| \frac{1}{t^2}\left(f(a + s u + t v) - f(a + s u) - f(a + t v) + f (a) \right) - (D^2 f)_a(u, v) \right\|_W\\
&\leq \frac{1}{|t|} \sup_{\tau \in (-|t|, |t|)} \left( \left\| \mathsf{Err}_{a}^{Df}(a + t u + \tau v) \right\|_{V \to W} 
+ \left\|\mathsf{Err}_a^{Df}(a + \tau v) \right\|_{V \to W} \right) \| v \|_V
\end{split}
\]
and observe that the limit of the right-hand side as $t\to 0$ is indeed $0$.
\end{proof}
