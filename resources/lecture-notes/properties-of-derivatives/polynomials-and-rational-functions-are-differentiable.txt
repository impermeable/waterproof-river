\begin{proposition}[Every polynomial is differentiable]
	Every polynomial on $\mathbb{R}^d$ is differentiable. 
\end{proposition}

\begin{proposition}[Every rational function is differentiable on its domain]
	Let $p : \mathbb{R}^d \to \mathbb{R}$ and $q : \mathbb{R}^d \to \mathbb{R}$ be two polynomials. Let 
	\[
	D := \{x \in \mathbb{R}^d \ | \ q(x) \neq 0 \}.
	\]
	Then the function $f: D \to \mathbb{R}$ given by
	\[
	f(x) = \frac{p(x)}{q(x)}
	\]
	is differentiable.
	
	In other words, every rational function is differentiable on its domain of definition.
\end{proposition}
