\begin{theorem}[Fundamental theorem of calculus]
\begin{enumerate}
\item
\label{it:fund-thm-calc-1} Let $f : [a, b] \to \mathbb{R}$ be continuous. Then the function
	\[
	F :[a, b] \to \mathbb{R}
	\]	
	given by
	\[
	F(x) := \int_a^x f(s) d s
	\]
	is differentiable on $(a, b)$ and for all $x \in (a, b)$
	\[
	F'(x) = f(x).
	\]
\item
\label{it:fund-thm-calc-2}
Let $F : [a, b] \to \mathbb{R}$ be continuous and let it be an anti-derivative of a function $f : [a, b] \to \mathbb{R}$, i.e. for all $x \in (a, b)$, 
\[
F'(x) = f(x)
\]
and suppose that $f$ is bounded and Riemann integrable on $[a, b]$. 

Then
\[
\int_a^b f(x) dx = F(b) - F(a).
\]

\end{enumerate}
\end{theorem}

\begin{proof}
We will first show part (\ref{it:fund-thm-calc-1}).	
	
We will show that the function $F$ is differentiable on $(a, b)$ and satisfies for all $c \in (a, b)$
\[
F'(c) = f(c).
\]	
To do so, we define $\mathsf{Err}_c: (a, b) \to \mathbb{R}$ by
\[
\mathsf{Err}_c(x) := F(x) - (F(c) + f(c) (x - c))
\]
and we need to show that
\[
\lim_{x \to c} \frac{\mathsf{Err}_c(x)}{|x - c|} = 0.
\]
Let $\epsilon > 0$. Because $f$ is continuous in $c$, there exists a $\delta > 0$ such that for all $x \in \mathbb(a, b)$, it holds that if $0 < |x - c| < \delta$ then
\[
|f(x) - f(c)| < \frac{\epsilon}{2}.
\]
Choose such a $\delta$.

Let $x \in (a, b)$ and assume $0  < | x -c | < \delta$. 
We have that
\[
\begin{split}
|\mathsf{Err}_c(x)|
&= |F(x) - F(c) - f(c) (x - c)|\\
&= \left|\int_c^x f(s) d s - f(c) (x - c)\right|\\
&= \left|\int_c^x (f(s) - f(c) dx \right|\\
&\leq \int_c^x |f(s) - f(c)| d x \\
&\leq \frac{\epsilon}{2} |x -c|
\end{split}
\]
Therefore
\[
\frac{| \mathsf{Err}_c(x)|}{|x - c|}< \epsilon.
\]

We will now show part (\ref{it:fund-thm-calc-2}) of the theorem.

Let $F : [a, b] \to \mathbb{R}$ be an anti-derivative of a function $f: [a, b] \to \mathbb{R}$ and assume that $f$ is bounded and Riemann integrable. 

We need to show that
\[
\int_a^b f(x) dx = F(b) - F(a).
\]
Let $P := \{x_0, x_1, \dots, x_n\}$ be a partition of $[a, b]$. We can write $F(b) - F(a)$ as a telescoping sum
\[
\begin{split}
F(b) - F(a)
&= \sum_{k=1}^n \big(F(x_k) - F(x_{k-1})\big)
\end{split}
\]
and then use the Mean-Value Theorem to conclude that there are points $c_k \in (x_{k-1}, x_k)$ such that
\[
\begin{split}
	F(b) - F(a)
	&= \sum_{k=1}^n \big(F(x_k) - F(x_{k-1})\big)\\
	&= \sum_{k=1}^n f(c_k) (x_k - x_{k-1})
\end{split}
\]
We now use the definition of 
\[
M_k := \sup_{x \in [x_{k-1}, x_k]} f(x)
\]
to estimate
\[
\begin{split}
		F(b) - F(a)
	&= \sum_{k=1}^n f(c_k) (x_k - x_{k-1})\\
	&\leq \sum_{k=1}^n M_k \Delta x_k\\
	&= U(P, f).
\end{split}
\]
It follows that
\[
F(b)- F(a) \leq \overline{\int_a^b} f(x) dx
\]
Similarly, we can prove that
\[
\underline{\int_a^b} f(x) dx \leq F(b) - F(a)
\]
Therefore,
\[
\underline{\int_a^b} f(x) dx 
\leq F(b) - F(a)
\leq \overline{\int_a^b} f(x) dx
\]
Because $f$ is integrable the left- and right-hand side of this inequality have the same value so that we can conclude that
\[
F(b) - F(a) = \int_a^b f(x) dx.
\]
\end{proof}
